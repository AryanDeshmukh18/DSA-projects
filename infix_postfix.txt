
#include <iostream>
#include <cstring>
#include <cctype>
using namespace std;

struct Node
{
    char data;
    Node *next;
};

class Stack
{
    Node *top;

public:
    Stack() { top = NULL; }

    void push(char ch)
    {
        Node *n = new Node;
        n->data = ch;
        n->next = top;
        top = n;
    }

    char pop()
    {
        if (!top) return '\0';

        Node *t = top;
        char x = t->data;
        top = top->next;
        delete t;
        return x;
    }

    char peek() { return top ? top->data : '\0'; }
    bool isEmpty() { return top == NULL; }
};

int precedence(char op)
{
    if (op == '^') return 3;
    if (op == '*' || op == '/') return 2;
    if (op == '+' || op == '-') return 1;
    return 0;
}

bool isOperator(char ch)
{
    return ch == '+' || ch == '-' || ch == '*' ||
           ch == '/' || ch == '^';
}

void infixToPostfix(char infix[], char postfix[])
{
    Stack s;
    int i = 0, j = 0;

    while (infix[i])
    {
        char ch = infix[i++];

        if (isalnum(ch))
            postfix[j++] = ch;

        else if (ch == '(')
            s.push(ch);

        else if (ch == ')')
        {
            while (!s.isEmpty() && s.peek() != '(')
                postfix[j++] = s.pop();

            if (!s.isEmpty()) s.pop();
        }

        else if (isOperator(ch))
        {
            while (!s.isEmpty() && s.peek() != '(' &&
                  (precedence(s.peek()) > precedence(ch) ||
                  (precedence(s.peek()) == precedence(ch) && ch != '^')))
                postfix[j++] = s.pop();

            s.push(ch);
        }
    }

    while (!s.isEmpty())
        if (s.peek() != '(')
            postfix[j++] = s.pop();
        else
            s.pop();

    postfix[j] = '\0';
}

void reverseString(char str[])
{
    int i = 0, j = strlen(str) - 1;

    while (i < j)
    {
        char t = str[i];
        str[i++] = str[j];
        str[j--] = t;
    }
}

void infixToPrefix(char infix[], char prefix[])
{
    char temp[100], postfix[100];

    strcpy(temp, infix);
    reverseString(temp);

    for (int i = 0; temp[i]; i++)
    {
        if (temp[i] == '(') temp[i] = ')';
        else if (temp[i] == ')') temp[i] = '(';
    }

    infixToPostfix(temp, postfix);
    strcpy(prefix, postfix);
    reverseString(prefix);
}

int main()
{
    char infix[100], postfix[100], prefix[100];

    cout << "Enter infix expression: ";
    cin >> infix;

    infixToPostfix(infix, postfix);
    infixToPrefix(infix, prefix);

    cout << "\nInfix expression: " << infix << endl;
    cout << "Postfix expression: " << postfix << endl;
    cout << "Prefix expression: " << prefix << endl;

    return 0;
}
